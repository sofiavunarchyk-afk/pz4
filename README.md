## Тема роботи
Опрацювання структури даних Стек (Stack) у мові програмування C++ за допомогою стандартного контейнера `std::stack`.

## Мета роботи
Набути практичних навичок роботи з контейнером `std::stack`, навчитися додавати елементи до стека (`push`), вилучати їх за принципом LIFO (Last In, First Out — «останній прийшов — перший вийшов»), отримувати доступ до верхнього елемента (`top`), а також перевіряти стек на порожнечу (`empty`).

### Завдання 1. Підрахунок символів у стеку рядків
**Програмний код:**
```cpp
#include <iostream>
#include <stack>
#include <string>
using namespace std;
int main() {
    stack<string> S; // Стек для зберігання рядків
    int count = 0; // Змінна для підрахунку символів

    // Додаємо рядки до стека
    S.push("apple");
    S.push("banana");
    S.push("cherry");
    // Виведіть усі елементи стека та підрахуйте символи
    while (!S.empty()) {
        string str = S.top(); // Отримати верхній елемент
        count += str.length(); // Додати кількість символів
        cout << str << endl; // Вивести рядок
        S.pop(); // Видалити верхній елемент
    }
    cout << "Загальна кількість символів: " << count << endl;
    return 0;
}
```
**Результат виконання:**
```text
cherry
banana
apple
Загальна кількість символів: 17
```

### Завдання 2. Додавання та вилучення цілих чисел, обчислення суми
**Програмний код:**
```cpp
#include <iostream>
#include <stack>
using namespace std;
int main() {
    stack<int> S; // Стек для зберігання цілих чисел
    int sum = 0; // Змінна для підрахунку суми
    int n; // Змінна для розміру стека
    // Запитати у користувача розмір стека
    cout << "Введіть розмір стека: ";
    cin >> n;
    // Додати елементи до стека
    cout << "Введіть " << n << " цілих чисел:" << endl;
    for (int i = 0; i < n; i++) {
        int num;
        cin >> num; // Ввести число
        S.push(num); // Додати число до стека
    }

    // Витягніть всі числа з стека та обчисліть суму
    cout << "Елементи стека: ";
    while (!S.empty()) {
        int num = S.top(); // Отримати верхній елемент
        cout << num << " "; // Вивести число
        sum += num; // Додати до суми
        S.pop(); // Видалити верхній елемент
    }
    
    cout << "\nСума чисел: " << sum << endl;
    
    return 0;
}
```

### Завдання 3. Пошук заданого елемента у стеку
**Програмний код:**
```cpp
#include <iostream>
#include <stack>
using namespace std;

int main() {
    stack<int> S; // Стек для зберігання цілих чисел
    int count = 0; // Змінна для підрахунку витягнутих елементів
    int foundCount = 0; // Змінна для підрахунку знайдених елементів

    // Додаємо елементи до стека
    S.push(10);
    S.push(20);
    S.push(30);
    S.push(40);
    S.push(50);
    S.push(30); // Додаємо повторюване число
    S.push(30); // Додаємо ще одне повторюване число

    int findNum;
    cout << "Введіть число для пошуку: ";
    cin >> findNum;
    
    // Витягніть елементи зі стека
    cout << "Витягнуті елементи з стека: ";
    while (!S.empty())
    {
        int num = S.top(); // Отримати верхній елемент
        S.pop(); // Видалити верхній елемент
        cout << num << " "; // Вивести число
        count++; // Збільшити лічильник

        if (num == findNum)
        {
            foundCount++; // Збільшити лічильник знайдених елементів
        }
    }
    cout << endl;

    // Виведення результатів
    if (foundCount > 0) {
        cout << "Число " << findNum << " знайдено " << foundCount << " раз(и)!" << endl;
    } else {
        cout << "Число " << findNum << " не знайдено." << endl;
    }

    cout << "Кількість витягнутих елементів: " << count << endl;

    return 0;
}
```
**Результат виконання:**
```text

Витягнуті елементи з стека: 30 30 50 40 30 20 10 
Число 30 знайдено 3 раз(и)!
Кількість витягнутих елементів: 7
```

### Завдання 4. Робота зі стеком випадкових чисел та фільтрація
**Програмний код:**
```cpp
#include <iostream>
#include <stack>
#include <cstdlib>  // Для rand() та srand()
#include <ctime>    // Для time()
using namespace std;

int main() {
    stack<int> S;
    const int size = 10; // Максимальна місткість стека
    int sum = 0; // Змінна для підрахунку суми парних чисел
    int count = 0; // Змінна для підрахунку кількості парних чисел
    
    srand((time(0))); // Ініціалізація генератора випадкових чисел
    
    // Генерація випадкових чисел у діапазоні від 1 до 20
    for (int i = 0; i < size; i++) {
        int randomNum = 1+rand() % 20; 
        S.push(randomNum);
    }
    
    // Виведіть усі парні числа з діапазону від 10 до 20 та обчисліть їх середнє значення
    cout << "Елементи стека (випадкові числа): " << endl;
    while (!S.empty())
    {
        int num = S.top(); // Отримати верхній елемент
        S.pop(); // Видалити верхній елемент
        
        // Вивести число
        cout << num << " ";
        
        // Перевірка на парність та діапазон
        if (num >= 10 && num <= 20 && num % 2 == 0)
        {
            sum += num; // Додати до суми
            count++; // Збільшити лічильник парних чисел
        }
    }
    cout << endl; // Перехід на новий рядок після виведення чисел
    
    // Обчислення та виведення середнього значення парних чисел
    if (count > 0) {
        cout << "Середнє значення парних чисел: " << double(sum) / count << endl;
    } else {
        cout << "Парних чисел не знайдено." << endl;
    }
    
    return 0;
}
```

### Завдання 5. Перевірка рядка на паліндром за допомогою стека
**Програмний код:**
```cpp
#include <iostream>
#include <string>
#include <stack>
using namespace std;
int main() {
    string word = "radar";
    stack<char> s;
    // Додавання кожної літери слова у стек
    for (char c : word) {
        s.push(c);
    }
    string reversedWord = "";

    // Витягнути літери у зворотному порядку
    while (!s.empty()) {
        reversedWord += s.top();
        s.pop();
    }
    cout << "Оригінал: " << word << endl;
    cout << "Навпаки:  " << reversedWord << endl;

    if (word == reversedWord) {
        cout << "Результат: Рядок є паліндромом!" << endl;
    } else {
        cout << "Результат: Рядок НЕ є паліндромом." << endl;
    }
    return 0;
}
```
**Результат виконання:**

```text
Результат: Рядок є паліндромом!
```
